// TAREFA: disparar-treino-custom — dispara treinar_modelo_custom.yml
// passando o config_id como input. Exige autenticação (mesmo mecanismo
// de disparar-predicoes/disparar-backtest).
async function tarefaDispararTreinoCustom(supabase, authHeader, configId) {
  if (!configId) return { status: 400, error: 'Parâmetro config_id é obrigatório.' };
  const resultado = await dispararWorkflow(supabase, authHeader, GITHUB_WORKFLOW_FILE_CUSTOM_TREINO, { config_id: configId });
  if (resultado.status === 200) {
    // Marca como aguardando_treino antes do workflow assumir (ele muda pra 'treinando')
    await supabase.table('custom_model_configs')
      .update({ status: 'aguardando_treino' })
      .eq('id', configId);
  }
  return resultado;
}
